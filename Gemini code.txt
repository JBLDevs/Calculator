// Keep track of the running total outside the event listener if needed, 
// or manage it dynamically like this:

add.addEventListener('click', function(Add) { 
    Add.preventDefault(); 
    
    // 1. If Display is empty, don't do anything
    if (Display.value === '') return;

    const currentInput = Number(Display.value);
    const newOperator = Add.target.value;

    // 2. Check if a first number already exists in the top display
    const existingFnumber = document.querySelector('.fnumber');

    if (!existingFnumber) {
        // FIRST CLICK: Just move the number to the top display
        firstNumber = currentInput;
        operator = newOperator;
        
        topdisplay.innerHTML = "<div class='fnumber'></div>";
        document.querySelector('.fnumber').textContent = `${firstNumber} ${operator}`;
    } else {
        // SUBSEQUENT CLICKS: Calculate the running total first
        const secondNumber = currentInput;
        
        if (operator === '+') {
            firstNumber = firstNumber + secondNumber; // Update the running total
        }
        
        // Update the operator to the newly clicked one
        operator = newOperator; 
        existingFnumber.textContent = `${firstNumber} ${operator}`;
    }

    // 3. Always clear the main display for the next input
    Display.value = ''; 
});
